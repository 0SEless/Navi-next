# M3.3.2 — Autosave Scheduler Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Create AutosaveSchedulerService that owns the timing of autosaves — when to call `WorkflowService.save('autosave')`, never how.

**Architecture:** New service subscribes to `eventBus` `document.changed` events, resets a debounce timer on each commit, and fires a max-interval safety net. Guards on `isDirty()` and `saveState === 'idle'` prevent unnecessary or concurrent saves.

**Tech Stack:** TypeScript, BaseEditorService pattern, existing eventBus/document.changed

## Global Constraints

- AutosaveScheduler never decides HOW to save — only WHEN to call `WorkflowService.save('autosave')`.
- No direct calls to `PersistenceService.save()` — always through `WorkflowService`.
- No subscription to `DocumentStore` directly — always through `eventBus` `document.changed`.
- The scheduler must never initiate concurrent saves — `saveState === 'idle'` guard prevents this.
- Timer values are constructor-configurable (for testing), never user-configurable.
- No UI awareness — no status bar, no toolbar, no settings.
- No start/stop/pause/resume methods — only init/destroy lifecycle.

```ts
import { BaseEditorService } from '../context'
import type { EditorServiceContext } from '../context/service-registry'
import type { WorkflowService } from './workflow-service'
import type { WorkflowStore, SaveState } from './workflow-store'
import type { DocumentEventBus } from '../eventbus'

export interface AutosaveSchedulerOptions {
  debounceMs?: number
  maxIntervalMs?: number
}

export class AutosaveSchedulerService extends BaseEditorService {
  readonly id = 'autosave'
  readonly dependencies = ['workflow', 'workflowStore', 'eventBus'] as const

  private debounceMs: number
  private maxIntervalMs: number
  private workflow!: WorkflowService
  private workflowStore!: WorkflowStore
  private eventBus!: DocumentEventBus

  private debounceTimer: ReturnType<typeof setTimeout> | null = null
  private maxIntervalTimer: ReturnType<typeof setInterval> | null = null
  private onDocumentChanged!: () => void

  constructor(options?: AutosaveSchedulerOptions) {
    super()
    this.debounceMs = options?.debounceMs ?? 5000
    this.maxIntervalMs = options?.maxIntervalMs ?? 30000
  }

  async init(context: EditorServiceContext): Promise<void> {
    await super.init(context)
    this.workflow = context.get('workflow')
    this.workflowStore = context.get('workflowStore')
    this.eventBus = context.get('eventBus')

    this.onDocumentChanged = () => this.handleDocumentChanged()
    this.eventBus.on('document.changed', this.onDocumentChanged)

    this.maxIntervalTimer = setInterval(() => this.tryAutosave(), this.maxIntervalMs)
  }

  async destroy(): Promise<void> {
    this.clearDebounce()
    if (this.maxIntervalTimer !== null) {
      clearInterval(this.maxIntervalTimer)
      this.maxIntervalTimer = null
    }
    if (this.onDocumentChanged) {
      this.eventBus.off('document.changed', this.onDocumentChanged)
    }
    await super.destroy()
  }

  private handleDocumentChanged(): void {
    this.clearDebounce()
    this.debounceTimer = setTimeout(() => this.tryAutosave(), this.debounceMs)
  }

  private tryAutosave(): void {
    if (!this.workflow) return
    if (!this.workflow.isDirty()) return
    if (this.workflowStore.getSnapshot().saveState !== 'idle') return

    this.workflow.save('autosave')
  }

  private clearDebounce(): void {
    if (this.debounceTimer !== null) {
      clearTimeout(this.debounceTimer)
      this.debounceTimer = null
    }
  }
}
```

```ts
import { describe, it, expect, vi, beforeEach, afterEach } from 'vitest'
import { AutosaveSchedulerService } from '../autosave-scheduler'
import { WorkflowStore } from '../workflow-store'
import { WorkflowService } from '../workflow-service'
import { DocumentEventBus } from '../../eventbus'
import type { EditorServiceContext } from '../../context/service-registry'

describe('AutosaveSchedulerService', () => {
  let scheduler: AutosaveSchedulerService
  let workflow: WorkflowService
  let workflowStore: WorkflowStore
  let eventBus: DocumentEventBus
  let context: EditorServiceContext
  let saveSpy: ReturnType<typeof vi.spyOn>

  beforeEach(async () => {
    eventBus = new DocumentEventBus()
    workflowStore = new WorkflowStore()

    workflow = { isDirty: vi.fn(), save: vi.fn() } as unknown as WorkflowService
    saveSpy = vi.spyOn(workflow, 'save')

    workflowStore.updateLifecycle({ saveState: 'saved', lastSavedAt: Date.now() })

    const store = new Map<string, any>()
    store.set('workflow', workflow)
    store.set('workflowStore', workflowStore)
    store.set('eventBus', eventBus)

    context = {
      get: (id: string) => store.get(id),
      document: {},
    } as unknown as EditorServiceContext

    scheduler = new AutosaveSchedulerService({ debounceMs: 50, maxIntervalMs: 5000 })
    await scheduler.init(context)
  })

  afterEach(async () => {
    await scheduler.destroy()
  })

  it('calls save after debounce delay when document is dirty and idle', async () => {
    vi.mocked(workflow.isDirty).mockReturnValue(true)

    eventBus.emit('document.changed', { version: 1 })

    await vi.waitFor(() => {
      expect(saveSpy).toHaveBeenCalledWith('autosave')
    }, { timeout: 200 })
  })

  it('resets debounce on rapid document changes', async () => {
    vi.mocked(workflow.isDirty).mockReturnValue(true)

    eventBus.emit('document.changed', { version: 1 })
    await vi.waitFor(() => {
      // First tick should not fire yet because debounce resets
      const calls = saveSpy.mock.calls.length
      expect(calls).toBe(0)
    }, { timeout: 30 })

    eventBus.emit('document.changed', { version: 2 })
    await vi.waitFor(() => {
      expect(saveSpy).toHaveBeenCalledWith('autosave')
    }, { timeout: 200 })

    // Should only fire once
    expect(saveSpy).toHaveBeenCalledTimes(1)
  })

  it('does not save if document is not dirty', async () => {
    vi.mocked(workflow.isDirty).mockReturnValue(false)

    eventBus.emit('document.changed', { version: 1 })

    await vi.waitFor(() => {
      expect(saveSpy).not.toHaveBeenCalled()
    }, { timeout: 200 })
  })

  it('does not save if saveState is not idle', async () => {
    vi.mocked(workflow.isDirty).mockReturnValue(true)
    workflowStore.updateLifecycle({ saveState: 'saving' })

    eventBus.emit('document.changed', { version: 1 })

    await vi.waitFor(() => {
      expect(saveSpy).not.toHaveBeenCalled()
    }, { timeout: 200 })
  })

  it('clears timers on destroy', async () => {
    vi.mocked(workflow.isDirty).mockReturnValue(true)

    await scheduler.destroy()
    eventBus.emit('document.changed', { version: 1 })

    // Small wait to ensure no autosave fires after destroy
    await new Promise(r => setTimeout(r, 100))
    expect(saveSpy).not.toHaveBeenCalled()
  })
})
```
